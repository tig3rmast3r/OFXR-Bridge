cmake_minimum_required(VERSION 3.24)
project(OFXRBridgeNeural LANGUAGES CXX)
set(NR_SDK_INCLUDE "" CACHE PATH "NGX SDK include folder")
add_library(xrfg_neural_rendering STATIC nr_feature.cpp nr_host.cpp nr_codec.cpp)
target_include_directories(xrfg_neural_rendering PUBLIC ../../include "${NR_SDK_INCLUDE}" PRIVATE .)
target_compile_features(xrfg_neural_rendering PUBLIC cxx_std_20)
target_compile_definitions(xrfg_neural_rendering PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
target_compile_options(xrfg_neural_rendering PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(xrfg_neural_rendering PUBLIC d3d12 dxgi d3dcompiler advapi32)
add_library(xrfg_neural_forwarder SHARED forwarder/dlssnr_forwarder.cpp)
target_link_libraries(xrfg_neural_forwarder PRIVATE d3d12)
set_target_properties(xrfg_neural_forwarder PROPERTIES OUTPUT_NAME "nvngx.dll_dlssnr")
add_executable(neural_contract_tests ../../tests/neural_contract_tests.cpp)
target_include_directories(neural_contract_tests PRIVATE .)
target_compile_features(neural_contract_tests PRIVATE cxx_std_20)
enable_testing()
add_test(NAME neural_contract_tests COMMAND neural_contract_tests)
add_executable(neural_copyback_contract_tests ../../tests/neural_copyback_contract_tests.cpp)
target_link_libraries(neural_copyback_contract_tests PRIVATE xrfg_neural_rendering)
target_compile_definitions(neural_copyback_contract_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME neural_copyback_contract_tests COMMAND neural_copyback_contract_tests)
add_executable(neural_codec_gpu_tests ../../tests/neural_codec_gpu_tests.cpp)
target_include_directories(neural_codec_gpu_tests PRIVATE .)
target_link_libraries(neural_codec_gpu_tests PRIVATE xrfg_neural_rendering)
target_compile_definitions(neural_codec_gpu_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME neural_codec_gpu_tests COMMAND neural_codec_gpu_tests)
set(VULKAN_INCLUDE "" CACHE PATH "Vulkan headers for independent command-stream validation")
if(VULKAN_INCLUDE)
    add_library(xrfg_vulkan_commands STATIC ../ingress/vulkan_command_stream.cpp)
    target_include_directories(xrfg_vulkan_commands PUBLIC ../../include "${VULKAN_INCLUDE}")
    target_compile_features(xrfg_vulkan_commands PUBLIC cxx_std_20)
    target_compile_definitions(xrfg_vulkan_commands PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(xrfg_vulkan_commands PRIVATE /W4 /permissive- /EHsc)
    add_executable(vulkan_command_tests ../ingress/vulkan_command_tests.cpp)
    target_link_libraries(vulkan_command_tests PRIVATE xrfg_vulkan_commands)
    add_test(NAME vulkan_command_tests COMMAND vulkan_command_tests)
endif()
