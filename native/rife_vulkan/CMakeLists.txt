cmake_minimum_required(VERSION 3.24)
project(OFXRRifeVulkan LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The RIFE Vulkan worker requires Windows x64")
endif()
set(OFXR_VULKAN_DEPS "" CACHE PATH "Pinned dependencies prepared by prepare_rife_vulkan_deps.py")
if(NOT EXISTS "${OFXR_VULKAN_DEPS}/ncnn/src/command.h")
  message(FATAL_ERROR "Run tools/prepare_rife_vulkan_deps.py --deps <directory> first")
endif()
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
set(NCNN_SIMPLEVK ON CACHE BOOL "" FORCE)
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_INSTALL_SDK OFF CACHE BOOL "" FORCE)
set(NCNN_RUNTIME_CPU OFF CACHE BOOL "" FORCE)
set(NCNN_AVX OFF CACHE BOOL "" FORCE)
set(NCNN_AVX2 OFF CACHE BOOL "" FORCE)
set(NCNN_AVX512 OFF CACHE BOOL "" FORCE)
add_subdirectory("${OFXR_VULKAN_DEPS}/ncnn" ncnn)
add_executable(ofxr_rife_vulkan worker.cpp)
target_compile_features(ofxr_rife_vulkan PRIVATE cxx_std_20)
target_compile_definitions(ofxr_rife_vulkan PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
  _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR OFXR_NCNN_REV="00357e2f20ea2f8245276c2fa1e443e15a4f6bfa")
target_include_directories(ofxr_rife_vulkan PRIVATE "${OFXR_VULKAN_DEPS}" "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
target_link_libraries(ofxr_rife_vulkan PRIVATE ncnn d3d12 dxgi bcrypt)
target_compile_options(ofxr_rife_vulkan PRIVATE /W4 /EHsc /permissive-)
